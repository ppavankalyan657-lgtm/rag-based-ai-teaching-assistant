
import streamlit as st
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

st.title("RAG-Based AI Teaching Assistant")

st.write("Ask questions based on your study materials.")

notes = st.text_area(
    "Enter your study notes",
    "Python is a programming language.\n"
    "Data Science analyzes data.\n"
    "Machine Learning learns patterns from data.\n"
    "RAG retrieves relevant information from documents."
)

question = st.text_input("Enter your question")

if st.button("Find Answer"):

    documents = [
        line.strip()
        for line in notes.split("\n")
        if line.strip()
    ]

    if not documents:
        st.warning("Please enter your study notes.")

    elif not question.strip():
        st.warning("Please enter a question.")

    else:
        vectorizer = TfidfVectorizer()

        vectors = vectorizer.fit_transform(
            documents + [question]
        )

        scores = cosine_similarity(
            vectors[-1],
            vectors[:-1]
        )[0]

        best_index = scores.argmax()

        if scores[best_index] == 0:
            st.error("No relevant information found.")

        else:
            st.subheader("Answer")
            st.write(documents[best_index])

            st.write(
                "Similarity Score:",
                round(scores[best_index], 2)
            )
